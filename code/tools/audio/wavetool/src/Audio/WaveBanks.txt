{

	XMA,
	XMA2,
	ADPCM
};

{

	i32 BitDepth = 16;
	i32 SampleRate = 48000;
	[Range(1, 100)]
	i32 XMACompressionQuality = 70;
	bool XMAHighFrequencyFilter = false;
};
